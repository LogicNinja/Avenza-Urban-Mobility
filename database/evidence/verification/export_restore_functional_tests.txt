    Create Procedure: CREATE DEFINER=`root`@`localhost` PROCEDURE `GetIncidentsBySegment`(
    IN p_segment_id INT
)
BEGIN
    SELECT
        incident_id,
        segment_id,
        incident_type,
        severity,
        occurred_at,
        reported_at
    FROM incidents
    WHERE segment_id = p_segment_id
    ORDER BY occurred_at DESC;
END
